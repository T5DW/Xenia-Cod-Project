cmake_minimum_required(VERSION 3.16)
project(bo1live CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(SOURCES src/net.cpp src/server.cpp src/main.cpp)
if(WIN32)
	add_executable(bo1live WIN32 ${SOURCES} src/app.rc)
	target_compile_definitions(bo1live PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0601 NOMINMAX)
	target_link_libraries(bo1live PRIVATE ws2_32 iphlpapi comctl32 gdi32 shell32)
	if(MSVC)
		target_link_options(bo1live PRIVATE /ENTRY:wWinMainCRTStartup /MANIFEST:NO)
		set_property(TARGET bo1live PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
	else()
		target_link_options(bo1live PRIVATE -municode -static)
	endif()
else()
	add_executable(bo1live ${SOURCES})
	find_package(Threads REQUIRED)
	target_link_libraries(bo1live PRIVATE Threads::Threads)
endif()
